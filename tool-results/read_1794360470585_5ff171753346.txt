    34→/* 与其他动画同一套绿色 LUT */
    35→function packRGB(r: number, g: number, b: number) {
    36→  return ((0xff000000 | (b << 16) | (g << 8) | r) >>> 0) as number
    37→}
    38→function buildLut(): Uint32Array {
    39→  const lut = new Uint32Array(256)
    40→  const stops: { p: number; c: [number, number, number] }[] = [
    41→    { p: 0.0, c: [8, 13, 11] },
    42→    { p: 0.55, c: [20, 124, 88] },
    43→    { p: 0.88, c: [96, 226, 166] },
    44→    { p: 1.0, c: [228, 255, 240] },
    45→  ]
    46→  for (let i = 0; i < 256; i++) {
    47→    const v = i / 255
    48→    let s0 = stops[0]
    49→    let s1 = stops[stops.length - 1]
    50→    for (let k = 0; k < stops.length - 1; k++) {
    51→      if (v >= stops[k].p && v <= stops[k + 1].p) {
    52→        s0 = stops[k]
    53→        s1 = stops[k + 1]
    54→        break
    55→      }
    56→    }
    57→    const f = (v - s0.p) / Math.max(1e-6, s1.p - s0.p)
    58→    lut[i] = packRGB(
    59→      Math.round(s0.c[0] + (s1.c[0] - s0.c[0]) * f),
    60→      Math.round(s0.c[1] + (s1.c[1] - s0.c[1]) * f),
    61→      Math.round(s0.c[2] + (s1.c[2] - s0.c[2]) * f)
    62→    )
    63→  }
    64→  return lut
    65→}
    66→const LUT = buildLut()
