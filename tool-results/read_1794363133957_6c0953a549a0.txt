    45→const shellVert = `
    46→#include <clipping_planes_pars_vertex>
    47→varying vec3 vN;
    48→varying vec3 vW;
    49→void main() {
    50→  vec4 wp = modelMatrix * vec4(position, 1.0);
    51→  vW = wp.xyz;
    52→  vN = normalize(mat3(modelMatrix) * normal);
    53→  vec4 mvPosition = viewMatrix * wp;
    54→  #include <clipping_planes_vertex>
    55→  gl_Position = projectionMatrix * mvPosition;
    56→}`
    57→
    58→const shellFrag = `
    59→#include <clipping_planes_pars_fragment>
    60→uniform vec3 uColor;
    61→uniform float uAlpha;
    62→uniform vec3 uCam;
    63→varying vec3 vN;
    64→varying vec3 vW;
    65→void main() {
    66→  #include <clipping_planes_fragment>
    67→  vec3 V = normalize(uCam - vW);
    68→  float ndv = abs(dot(V, normalize(vN)));
    69→  float rim = pow(1.0 - ndv, 1.8);
    70→  float a = uAlpha * (0.12 + 1.5 * rim);
    71→  gl_FragColor = vec4(uColor, clamp(a, 0.0, 0.95));
    72→}`

   432→          fenceColArr[ti + 2] = m
   433→        }

   435→        topLinePos[i * 3 + 1] = y
   436→        topLinePos[i * 3 + 2] = 0
   437→      }

   439→      fenceColAttr.needsUpdate = true
   440→      topLineAttr.needsUpdate = true
   441→      /* 探针：固定半径处，高度 = 此刻场值（示波器画的正是它） */
   442→      const vp =
   443→        Math.cos(K3 * rp - OMEGA3 * t) * Math.min(1, 1.7 / Math.max(rp, 0.5))

   452→      guidePos[5] = 0

   484→    /* ---- 读数横幅 ---- */
   485→    const updateReadouts = (eP: number, rising: boolean, pausedNow: boolean) => {

   491→        let cls: string

   497→          txt = c.text
   498→          cls = TONE[c.kind] ?? TONE.rise
   499→        }
   500→        if (badge.textContent !== txt) {
   501→          badge.textContent = txt
   502→          badge.className = base + cls
   503→        }
